# CONSULTAS USUARIOS
# -------------------
# Ver todos los usuarios
SELECT id, nombre_usuario, email, rol, activo, fecha_creacion 
FROM auth.usuarios 
ORDER BY id;

# Ver solo usuarios activos
SELECT id, nombre_usuario, email, rol 
FROM auth.usuarios 
WHERE activo = TRUE;

# Ver solo administradores
SELECT id, nombre_usuario, email 
FROM auth.usuarios 
WHERE rol = 'administrador' AND activo = TRUE;

# Buscar usuario por nombre
SELECT * FROM auth.usuarios 
WHERE nombre_usuario = 'admin';

# Buscar usuario por email
SELECT * FROM auth.usuarios 
WHERE email LIKE '%@evolve.cl';

# Contar usuarios por rol
SELECT rol, COUNT(*) as total 
FROM auth.usuarios 
WHERE activo = TRUE 
GROUP BY rol;


# CONSULTAS SESIONES
# -------------------
# Ver sesiones activas con información de usuario
SELECT 
    s.id,
    u.nombre_usuario,
    u.email,
    s.direccion_ip,
    s.user_agent,
    s.fecha_creacion,
    s.fecha_ultima_actividad,
    EXTRACT(EPOCH FROM (NOW() - s.fecha_ultima_actividad))/60 as minutos_inactivo
FROM auth.sesiones_usuario s
JOIN auth.usuarios u ON s.id_usuario = u.id
WHERE s.activa = TRUE
ORDER BY s.fecha_ultima_actividad DESC;

# Ver todas las sesiones de un usuario específico
SELECT 
    id,
    token_sesion,
    direccion_ip,
    fecha_creacion,
    fecha_ultima_actividad,
    activa
FROM auth.sesiones_usuario
WHERE id_usuario = 1
ORDER BY fecha_ultima_actividad DESC;

# Contar sesiones activas
SELECT COUNT(*) as sesiones_activas 
FROM auth.sesiones_usuario 
WHERE activa = TRUE;

# Ver sesiones expiradas (más de 24 horas de inactividad)
SELECT 
    u.nombre_usuario,
    s.direccion_ip,
    s.fecha_ultima_actividad
FROM auth.sesiones_usuario s
JOIN auth.usuarios u ON s.id_usuario = u.id
WHERE s.activa = TRUE 
AND s.fecha_ultima_actividad < NOW() - INTERVAL '24 hours'
ORDER BY s.fecha_ultima_actividad;


# CONSULTAS INTENTOS DE LOGIN
# ----------------------------
# Ver últimos intentos de login
SELECT 
    nombre_usuario,
    direccion_ip,
    exitoso,
    fecha_intento,
    mensaje
FROM auth.intentos_login
ORDER BY fecha_intento DESC
LIMIT 50;

# Ver intentos fallidos
SELECT 
    nombre_usuario,
    direccion_ip,
    COUNT(*) as intentos_fallidos,
    MAX(fecha_intento) as ultimo_intento
FROM auth.intentos_login
WHERE exitoso = FALSE
GROUP BY nombre_usuario, direccion_ip
HAVING COUNT(*) >= 3
ORDER BY intentos_fallidos DESC;

# Ver intentos desde una IP específica
SELECT *
FROM auth.intentos_login
WHERE direccion_ip = '192.168.1.100'
ORDER BY fecha_intento DESC;

# Estadísticas de login del día
SELECT 
    exitoso,
    COUNT(*) as total,
    COUNT(DISTINCT nombre_usuario) as usuarios_unicos,
    COUNT(DISTINCT direccion_ip) as ips_unicas
FROM auth.intentos_login
WHERE fecha_intento >= CURRENT_DATE
GROUP BY exitoso;


# OPERACIONES DE USUARIOS

# Activar/Desactivar usuario
UPDATE auth.usuarios 
SET activo = FALSE 
WHERE id = 1;

UPDATE auth.usuarios 
SET activo = TRUE 
WHERE id = 1;

# Cambiar rol de usuario
UPDATE auth.usuarios 
SET rol = 'administrador' 
WHERE id = 2;

# Actualizar email
UPDATE auth.usuarios 
SET email = 'nuevo@email.com' 
WHERE id = 1;

# Actualizar contraseña (hash generado desde Python)
UPDATE auth.usuarios 
SET hash_contraseña = 'nuevo-hash-aqui' 
WHERE nombre_usuario = 'admin';

# Eliminar usuario (soft delete)
UPDATE auth.usuarios 
SET activo = FALSE 
WHERE id = 1;


# OPERACIONES DE SESIONES
# ------------------------
# Cerrar sesión específica
UPDATE auth.sesiones_usuario 
SET activa = FALSE 
WHERE token_sesion = 'token-aqui';

# Cerrar todas las sesiones de un usuario
UPDATE auth.sesiones_usuario 
SET activa = FALSE 
WHERE id_usuario = 1;

# Cerrar sesiones expiradas (usando función)
SELECT auth.limpiar_sesiones_expiradas(1440); -- 24 horas

# Cerrar sesiones expiradas (manual)
UPDATE auth.sesiones_usuario 
SET activa = FALSE 
WHERE fecha_ultima_actividad < NOW() - INTERVAL '24 hours' 
AND activa = TRUE;

# Eliminar sesiones antiguas permanentemente
DELETE FROM auth.sesiones_usuario 
WHERE fecha_ultima_actividad < NOW() - INTERVAL '30 days';

# Crear usuario de aplicación

    tc.constraint_name,
    tc.table_name,
    kcu.column_name,
    tc.constraint_type
FROM information_schema.table_constraints tc
JOIN information_schema.key_column_usage kcu

WHERE tc.table_schema = 'auth'
ORDER BY tc.table_name, tc.constraint_type;

# LOGS Y AUDITORÍA
# ----------------
# Ver últimas actividades por usuario
SELECT 
    u.nombre_usuario,
    MAX(s.fecha_ultima_actividad) as ultima_actividad,
    COUNT(DISTINCT s.id) as sesiones_totales,
    COUNT(DISTINCT s.direccion_ip) as ips_diferentes
FROM auth.usuarios u
LEFT JOIN auth.sesiones_usuario s ON u.id = s.id_usuario
WHERE u.activo = TRUE
GROUP BY u.nombre_usuario
ORDER BY ultima_actividad DESC NULLS LAST;

# Ver usuarios nunca conectados
SELECT nombre_usuario, email, fecha_creacion
FROM auth.usuarios
WHERE fecha_ultimo_acceso IS NULL
AND activo = TRUE;
